1. Summary

Scope of verification. The Okta, Auth0 and Microsoft Entra ID adapters were tested against an in-process fake identity provider (pkg/sso/oidc/oidctest), not against live tenant credentials. The unit tests and tests/e2e.sh prove that the adapters, the gateway and the five example clients work together over HTTP. They do not prove that a real Okta, Auth0 or Entra tenant accepts your configuration. A real sign-in needs your tenant, credentials and a browser, and follows the matching Docs/*-SSO-Setup.md guide.

Auth0: slash issuer, audience, roles, logout pkg/sso/auth0/provider.go:59

func New(cfg Config) (*oidc.Provider, error) {
	if cfg.ID == "" {
		cfg.ID = DriverName
	}
	if cfg.RolesClaim == "" {
		cfg.RolesClaim = "roles"
	}
	issuer := cfg.IssuerURL
	if issuer != "" && !strings.HasSuffix(issuer, "/") {
		issuer += "/" // Auth0 issuers always end with a slash
	}
	extra := map[string]string{}
	if cfg.Audience != "" {
		extra["audience"] = cfg.Audience
	}
	return oidc.New(oidc.Config{
		ID: cfg.ID, Name: "Auth0",
		ClientID: cfg.ClientID, ClientSecret: cfg.ClientSecret, RedirectURL: cfg.RedirectURL,
		Issuer: issuer, DiscoveryURL: cfg.DiscoveryURL, Scopes: cfg.Scopes,
		HTTPClient: cfg.HTTPClient, ExtraAuthParams: extra,
		MapUser: func(raw map[string]any, u *sso.UserInfo) {
			if v := oidc.String(raw, "nickname"); v != "" {
				u.Username = v
			}
			u.Roles = oidc.Strings(raw, cfg.RolesClaim)
			u.Groups = oidc.Strings(raw, "groups")
			u.TenantID = oidc.String(raw, "org_id")
		},
		// Auth0 does not use end_session_endpoint for its default tenant
		// domain: logout is <issuer>v2/logout?client_id=...&returnTo=...
		LogoutURL: func(req sso.LogoutRequest) (string, error) {
			q := url.Values{"client_id": {cfg.ClientID}}
			if req.PostLogoutURL != "" {
				q.Set("returnTo", req.PostLogoutURL)
			}
			return issuer + "v2/logout?" + q.Encode(), nil
		},
	})
}

Shared base: ID token verification pkg/sso/oidc/provider.go:265

RS256 only. Checks signature, issuer, audience, exp, nbf and subject. The nonce is checked in HandleCallback.

func (p *Provider) verifyIDToken(ctx context.Context, token string) (*parsedJWT, error) {
	d, err := p.endpoints(ctx)
	if err != nil {
		return nil, err
	}
	j, err := parseJWT(token)
	if err != nil {
		return nil, sso.ErrTokenValidation(p.errf("invalid id token"), err)
	}
	pub, err := p.key(ctx, d, j.header.Kid)
	if err != nil {
		return nil, err
	}
	if err := j.verify(pub); err != nil {
		return nil, sso.NewError(sso.ErrCodeSignatureInvalid, p.errf("id token signature invalid"), http.StatusUnauthorized, nil)
	}
	c, now := j.claims, p.now()
	switch {
	case !p.issuerOK(c.Issuer, d, j.raw):
		return nil, sso.ErrTokenValidation(p.errf("unexpected issuer"), nil)
	case !c.Audience.contains(p.cfg.ClientID):
		return nil, sso.ErrTokenValidation(p.errf("audience mismatch"), nil)
	case c.ExpiresAt == 0 || now.After(time.Unix(c.ExpiresAt, 0).Add(clockSkew)):
		return nil, sso.NewError(sso.ErrCodeTokenExpired, p.errf("id token expired"), http.StatusUnauthorized, nil)
	case c.NotBefore != 0 && now.Add(clockSkew).Before(time.Unix(c.NotBefore, 0)):
		return nil, sso.ErrTokenValidation(p.errf("id token not yet valid"), nil)
	case c.Subject == "":
		return nil, sso.ErrTokenValidation(p.errf("id token has no subject"), nil)
	}
	if p.cfg.CheckClaims != nil {
		if err := p.cfg.CheckClaims(j.raw); err != nil {
			return nil, err
		}
	}
	return j, nil
}
Shared base: algorithm allow-list pkg/sso/oidc/jwt.go:75

Rejects none, HS256 and every non-RS256 token before any key lookup.

func parseJWT(token string) (*parsedJWT, error) {
	parts := strings.Split(token, ".")
	if len(parts) != 3 {
		return nil, errMalformedJWT
	}
	dec := base64.RawURLEncoding
	hb, err := dec.DecodeString(parts[0])
	if err != nil {
		return nil, errMalformedJWT
	}
	pb, err := dec.DecodeString(parts[1])
	if err != nil {
		return nil, errMalformedJWT
	}
	sig, err := dec.DecodeString(parts[2])
	if err != nil {
		return nil, errMalformedJWT
	}
	j := &parsedJWT{signing: parts[0] + "." + parts[1], sig: sig}
	if err := json.Unmarshal(hb, &j.header); err != nil {
		return nil, errMalformedJWT
	}
	if err := json.Unmarshal(pb, &j.claims); err != nil {
		return nil, errMalformedJWT
	}
	if err := json.Unmarshal(pb, &j.raw); err != nil {
		return nil, errMalformedJWT
	}
	if j.header.Alg != "RS256" {
		return nil, errors.New("unsupported jwt algorithm: " + j.header.Alg)
	}
	return j, nil
}
Gateway: placeholder-aware vendor wiring cmd/ssod/vendors.go:20

A fresh .env.example keeps all three vendors off. A client id without a secret is an error.

func vendorEnv(getenv func(string) string, name, prefix string) (id, secret, redirect string, ok bool, err error) {
	id, secret = getenv(prefix+"_CLIENT_ID"), getenv(prefix+"_CLIENT_SECRET") // nosecret: env var names
	if isPlaceholder(id) || id == legacyOktaSampleID {
		return "", "", "", false, nil
	}
	if isPlaceholder(secret) {
		return "", "", "", false, fmt.Errorf("%s_CLIENT_SECRET is not set (client id is configured)", prefix) // nosecret
	}
	redirect = getenv(prefix + "_REDIRECT_URL")
	if redirect == "" {
		base := strings.TrimRight(getenv("SSO_SERVER_BASE_URL"), "/")
		if base == "" {
			return "", "", "", false, fmt.Errorf("set %s_REDIRECT_URL or SSO_SERVER_BASE_URL", prefix)
		}
		redirect = base + "/auth/" + name + "/callback"
	}
	return id, secret, redirect, true, nil
}
